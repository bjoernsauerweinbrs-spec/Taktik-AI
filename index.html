async function askToni() { const input = document.getElementById('user-input'); const text = input.value; if (!text) return; addMsg("user", text); input.value = ""; const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${USER_KEY}`; try { const response = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: `Du bist Toni, ein Fußballtrainer. Antworte kurz: ${text}` }] }] }) }); const data = await response.json(); // HIER IST DIE NEUE DIAGNOSE: if (data.error) { // Toni verrät uns jetzt den echten Grund von Google! addMsg("toni", "Google sagt: " + data.error.message + " (Fehler-Code: " + data.error.status + ")"); } else if (data.candidates && data.candidates[0].content.parts[0].text) { addMsg("toni", data.candidates[0].content.parts[0].text); } } catch (e) { addMsg("toni", "Verbindung zum Google-Server konnte nicht aufgebaut werden."); } }
