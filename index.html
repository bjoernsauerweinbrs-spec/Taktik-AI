async function askToni() { const input = document.getElementById('user-input'); const text = input.value; if (!text) return; addMsg("user", text); input.value = ""; // Wir rufen Google direkt auf const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${USER_KEY}`; try { const response = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: `Du bist Toni, ein Fußballtrainer für Kinder. Antworte kurz: ${text}` }] }] }) }); const data = await response.json(); // DIAGNOSE-MODUS: if (data.error) { // Das hier sagt uns jetzt, was wirklich los ist! addMsg("toni", "Google-Fehler: " + data.error.message + " (" + data.error.status + ")"); } else if (data.candidates && data.candidates[0].content.parts[0].text) { addMsg("toni", data.candidates[0].content.parts[0].text); } } catch (e) { addMsg("toni", "Technischer Fehler: Verbindung zum Server fehlgeschlagen."); } }
